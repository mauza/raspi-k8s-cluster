# ArgoCD Installation via Helm
# Version: v3.2.0 (Chart: argo-cd-9.1.0)
# Migrated from static manifests on 2025-11-09

# Prerequisites
# Add the ArgoCD Helm repository
helm repo add argo https://argoproj.github.io/argo-helm
helm repo update

# Installation
# Install ArgoCD using the values.yaml configuration
helm install argocd argo/argo-cd --version 9.1.0 --namespace argocd --values /home/mauza/Projects/raspi-k8s-cluster/argo/values.yaml

# Apply the ingress configuration
kubectl apply -f /home/mauza/Projects/raspi-k8s-cluster/argo/argo.yaml

# Verify installation
helm list -n argocd
kubectl get pods -n argocd
kubectl get applications -n argocd

# Upgrade ArgoCD
# Update the repo and upgrade to a newer version
helm repo update
helm search repo argo/argo-cd --versions
helm upgrade argocd argo/argo-cd --version <NEW_VERSION> --namespace argocd --values /home/mauza/Projects/raspi-k8s-cluster/argo/values.yaml

# Get admin password
kubectl -n argocd get secret argocd-initial-admin-secret -o jsonpath="{.data.password}" | base64 -d && echo

# Access the UI
# Via port-forward:
kubectl port-forward service/argocd-server -n argocd 8080:80
# Then visit: http://localhost:8080

# Via ingress:
# https://argo.mau.guru

# Uninstall (if needed)
# This will preserve CRDs and Application definitions
helm uninstall argocd -n argocd

# Rollback to previous version (if needed)
helm rollback argocd -n argocd

# View deployment history
helm history argocd -n argocd
